# TMNT (MAME set 'tmnt', World 4 Players, version X) - Z80 sound program symbols (963e20.g13)
#
# Addresses and names only: no game data lives in this file.
# Format:  ADDRESS  KIND  NAME  [...]   [# comment: how it was found]
#   func     a function entry (one nothing calls is decompiled too: code the entry points never reach)
#   loop     the main loop's top, a function entry: where the program waits for the board (runtime/src/sound_cpu.c
#            runs the main loop from there while the board's time is ahead)
#   code     a code label (a place the code jumps into the middle of a routine: set_channel_handler)
#   jumps    NAME COUNT [first=K]: a table of COUNT 16-bit code addresses, which the code indexes from
#            ADDRESS; with first=K it never reads the first K entries (the table starts K entries on)
#   data     NAME TYPE[COUNT] [TARGET]: COUNT values of TYPE (u8, u16, or a record: struct NAME SIZE rom);
#            TARGET, what each u16 points at (a record, or the elements a pointer walks)
#   codedata NAME TYPE[COUNT]: code the program also reads as data (the bytes stay code)
#   fill     COUNT: bytes nothing reads (none: the program's own fill is the 0xFF after its last track)
# Addresses are the Z80's: 0000-7FFF the program ROM, 8000-87FF the RAM, 9000 sres (bit 2 the title music
# sample), A000 the 68000's sound command (the latch), B000-B00D the K007232, C000-C001 the YM2151, D000 the
# uPD7759's sample number, E000 its start line, F000 its busy line.
#
# The analysis (tools/core/z80/analysis.py, game_profile's SOUND_ settings) starts at the reset (0000) and
# the interrupt (0038) and finds the code without traces, following what the registers hold into the jumps
# through them (z80/values.py): the main loop runs each channel through the handler its record keeps at +2
# (run_channel_handler, JP (HL) at 05CF), which the code stores there (set_channel_handler and the like),
# and the music's note hook at +28 (09FC). The jumps lines give the tables' lengths, which the code does not.
# The data lines describe the rest of the program for tools/sound_coverage.py; tools/sound_data_walk.py
# walks the music tracks the song table points at.
#
# Names for the decompiled C (tools/decompile_sound.py), as symbols.txt has them (disassemble.load_names):
#   struct NAME SIZE rom, field STRUCT OFFSET NAME TYPE[N] [TARGET]: a record of the program ROM and its
#   fields; elements NAME TYPE: values a pointer walks; param FUNCTION REGISTER NAME STRUCT: what a function
#   takes in a register (the sound coverage takes every constant the analysis sees there as one);
#   ADDRESS io NAME TYPE: a register of the board on the Z80's bus; ADDRESS var NAME[COUNT] TYPE: a variable in
#   the RAM; ADDRESS table NAME TYPE: a table the C reads by name (the game loads the asset piece of that name);
#   struct NAME SIZE, array STRUCT FIRST COUNT, field: records in the RAM.

# ======== the driver ========
#
# What the program is (read from the listing): the 68000 writes a sound command to the latch and raises the
# Z80's IRQ; sound_command (0038) takes it at once: 00 resets the sound, FF silences it, FE fades the music
# out, 01-41 start an FM effect, 60-78 a song, 7D and 7E start and stop the title music (the 512 KB sample
# MAME's tmnt.cpp plays through sres bit 2), 80-93 a PCM effect (the K007232), A0-C3 a speech (the
# uPD7759), DF stops the speech. The main loop waits for the YM2151's timer, then runs 15 channel records of
# 40h bytes (8000-83BF), each through the handler it keeps at +2: 8000 the FM effects that come first (YM2151
# channel 0), 8040-81C0 the music's 7 FM channels (1-7) and 8200, 8240 its 2 PCM channels (the K007232's A
# and B), 8280 and 82C0 the PCM effects (A, B), 8300 the speech, 8340 and 8380 the FM effects that take the
# music's channels 6 and 7. A record's +0 is its chip channel (bit 3: the K007232, bit 0 its channel B), +1
# the sound it plays, +F its priority.

# The 15 channel records (8000-83BF, 40h bytes each): 8000 the FM effects' first channel, 8040-81C0 the music's
# FM channels 1-7, 8200 and 8240 its PCM channels, 8280 and 82C0 the PCM effects', 8300 the speech's, 8340 and
# 8380 the FM effects' on the music's channels 6 and 7. Every handler runs on one in IX; the fields past +3
# mean one thing to the music and others to each effect (named in the naming rounds)
struct  channel 40
array   channel 8000 15
field   channel 0  chip_channel u8      # the YM2151's channel, or bit 3 the K007232 (bit 0 its channel B)
field   channel 1  sound u8             # the sound it plays (0: free)
field   channel 2  handler u16          # the routine that runs it on the next pass (run_channel_handler)
field   channel F  priority u8          # the priority of what it plays

field   pcm_drum 3 level u8
